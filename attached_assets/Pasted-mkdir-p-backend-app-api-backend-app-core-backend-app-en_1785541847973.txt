# إنشاء ملف .env.example
cat > .env.example << 'EOF'
DATABASE_URL=postgresql+asyncpg://trustion:trustion@localhost:5432/trustion
REDIS_URL=redis://localhost:6379
EOF

# docker-compose.yml
cat > docker-compose.yml << 'EOF'
version: "3.8"
services:
  api:
    build: ./backend
    environment:
      - DATABASE_URL=${DATABASE_URL}
      - REDIS_URL=${REDIS_URL}
      - EDGE_SIGNER_MODE=software
    depends_on:
      - db
      - redis
    command: sh -c "alembic upgrade head && python -m app.scripts.load_regulatory_data && uvicorn app.main:app --host 0.0.0.0 --port 8000"
  frontend:
    build: ./frontend
    ports:
      - "80:80"
  db:
    image: timescale/timescaledb:latest-pg15
    environment:
      - POSTGRES_USER=trustion
      - POSTGRES_PASSWORD=${POSTGRES_PASSWORD}
      - POSTGRES_DB=trustion
    volumes:
      - pgdata:/var/lib/postgresql/data
  redis:
    image: redis:7-alpine
volumes:
  pgdata:
EOF
